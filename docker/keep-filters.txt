# Filters kept next to gpufx; FFmpeg's own video filters are disabled, effects come from
# gpufx packages instead. The filters below stay because the ffmpeg CLI or gpufx needs
# them, or because they normalise the inputs without being a visual effect.
#   build.sh (full Docker build): this list plus every audio filter.
#   scripts/build-prebuilt.sh:    this list only. Any other filter (scale_cuda, drawtext...)
#                                 gives "No such filter"; add it here to keep it.

# Graph plumbing (buffer/buffersink are always built and need no entry)
null
format

# Audio plumbing the ffmpeg CLI inserts by itself for any file with sound:
# anull (no -af), aformat/aresample (format and rate conversion), atrim (-ss/-t), apad.
# The Docker build keeps every audio filter anyway; scripts/build-prebuilt.sh needs these.
anull
aformat
aresample
atrim
apad

# Pixel format conversion: gpufx works in RGBA, decoders output YUV.
scale

# Used by the ffmpeg CLI itself (-ss/-t trimming, autorotate, cropping metadata).
trim
crop
hflip
vflip
transpose
rotate

# Change gpufx params while running: sendcmd=c='2.0 gpufx@fx params strength=0.2'
sendcmd

# Feed one input to several gpufx chains: [0]split[a][b]
split

# Hardware frames: gpufx processes Vulkan frames in place (-init_hw_device vulkan,
# hwupload,gpufx=...,hwdownload,format=...); hwmap maps frames between devices.
# gpufx_upload/gpufx_download: pipelined replacements for hwupload/hwdownload (Vulkan).
hwupload
hwdownload
gpufx_upload
gpufx_download
hwmap

# Utility filters to normalise the inputs: timing, metadata and layout, and the few
# picture filters that only bring a stream to a standard form (deinterlace, colour space).
# Frame rate and timestamps
fps
setpts
asetpts
settb
asettb
# Aspect ratio and colour metadata
setsar
setdar
setrange
setparams
setfield
colorspace
# Deinterlacing
yadif
bwdif
# Padding, joining, picking and looping frames
pad
tpad
concat
select
aselect
loop
aloop
reverse
areverse
# Debugging and sinks
showinfo
ashowinfo
copy
acopy
nullsink
anullsink
# Basic audio: level, mixing, delay, fades, channels, tempo
volume
amix
amerge
adelay
afade
pan
atempo

# The effect filter
gpufx
